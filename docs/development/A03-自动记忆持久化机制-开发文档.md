## 一、需求背景与目标

### 1.1 问题陈述（P01）

**现状问题**: Memory系统和Agent管道已部分打通（A-02完成60%），但学习数据的持久化仍需手动触发或完全缺失。

**影响范围**:
- ❌ 用户解题历史无法自动积累
- ❌ 错题本数据依赖前端手动添加
- ❌ 用户画像无法动态更新
- ❌ 智能练习系统的推荐引擎无数据支撑

### 1.2 核心目标

✅ **自动化**: Agent完成解题后自动提取并持久化学习事件  
✅ **异步化**: 持久化不阻塞主流程（流式输出）  
✅ **可靠性**: 持久化失败仅记日志，不影响用户体验  
✅ **智能化**: 自动识别题目分类和难度等级  
✅ **批量优化**: 每10条事件或5秒批量写入，减少DB压力

#### 4.2.2 核心方法伪代码

##### 方法: `persist_interaction()`

```python
async def persist_interaction(
    self,
    user_id: str,
    session_id: str,
    user_input: str,
    assistant_response: str,
    context: Dict[str, Any],
    metadata: Optional[Dict] = None,
) -> bool:
    try:
        # 1. 提取学习事件
        event = await self._extract_learning_event(
            user_input=user_input,
            assistant_response=assistant_response,
            context=context,
        )
        
        if not event:
            logger.debug(f"无法从交互中提取学习事件: session={session_id}")
            return False
        
        # 2. 补充元数据
        event["user_id"] = user_id or "anonymous"
        event["metadata_"] = {
            **(event.get("metadata_", {}) or {}),
            "session_id": session_id,
            "persisted_at": datetime.now(timezone.utc).isoformat(),
            **(metadata or {}),
        }
        
        # 3. 加入缓冲区
        success = await self._add_to_buffer(event)
        
        if success:
            logger.debug(
                f"学习事件已入缓冲区: "
                f"user={user_id}, category={event.get('category')}, "
                f"difficulty={event.get('difficulty')}"
            )
        
        return success
        
    except Exception as e:
        logger.error(f"持久化交互失败: {e}", exc_info=True)
        return False
```

##### 方法: `_extract_learning_event()`

```python
async def _extract_learning_event(
    self,
    user_input: str,
    assistant_response: str,
    context: Dict[str, Any],
) -> Optional[Dict[str, Any]]:
    """
    从交互中提取学习事件的智能逻辑。
    
    提取策略：
    1. 检测是否是数学相关问题（排除闲聊）
    2. 从context中提取工具调用信息
    3. 识别题目内容和答案
    4. 推断是否是错题场景
    """
    
    # 1. 过滤非数学问题
    if not self._is_math_question(user_input):
        return None
    
    # 2. 基础字段提取
    event = {
        "event_type": EVENT_TYPE_QUESTION,
        "question_content": user_input[:2000],  # 截断过长题目
        "correct_answer": assistant_response[:5000],  # AI回答作为参考答案
        "created_at": datetime.now(timezone.utc),
    }
    
    # 3. 从上下文提取额外信息
    thought_recorder = context.get("thought_recorder")
    if thought_recorder:
        # 提取使用的工具
        tools_used = self._extract_tools_from_thought(thought_recorder)
        if tools_used:
            event["tools_used"] = ",".join(tools_used)
        
        # 提取解题步骤（用于难度估算）
        solution_steps = getattr(thought_recorder, 'get_steps', lambda: [])()
        if solution_steps:
            event["metadata_"] = event.get("metadata_", {}) or {}
            event["metadata_"]["solution_step_count"] = len(solution_steps)
    
    # 4. 自动分类
    category = await self._extract_category(
        question_text=user_input,
        response_text=assistant_response,
    )
    event["category"] = category
    
    # 5. 难度估算
    difficulty = await self._estimate_difficulty(
        question_text=user_input,
        solution_steps=event.get("metadata_", {}).get("solution_step_count"),
        time_spent=self._calculate_duration(context),
        tools_used=event.get("tools_used", "").split(",") if event.get("tools_used") else None,
    )
    event["difficulty"] = difficulty
    
    # 6. 检测是否可能是错题（基于用户后续行为）
    # 注意：真正的错题标记由persist_feedback()处理
    
    return event


def _is_math_question(self, text: str) -> bool:
    """快速判断是否是数学问题"""
    math_indicators = [
        r"[+\-*/=∫∑∏√∂∞≈≤≥]",  # 数学符号
        r"\b(sin|cos|tan|log|ln|lim)\b",  # 数学函数
        r"\b(求|计算|证明|解|推导)\b",  # 动词
        r"\d+\s*[+\-*/]\s*\d+",  # 算术表达式
        r"(f\(x\)|y\s*=|d/dx)",  # 函数表达式
    ]
    
    for pattern in math_indicators:
        if re.search(pattern, text, re.IGNORECASE):
            return True
    
    return len(text) > 5  # 较长的文本也可能是数学问题描述
```

##### 方法: `_estimate_difficulty()` (多因子加权)

```python
async def _estimate_difficulty(
    self,
    question_text: str,
    solution_steps: Optional[int] = None,
    time_spent: Optional[int] = None,
    tools_used: Optional[List[str]] = None,
) -> int:
    """
    多因子难度估算算法。
    
    因子权重：
    - 文本复杂度（长度+符号密度）：30%
    - 解题步骤数：25%
    - 耗时：20%
    - 工具使用：15%
    - 关键词高级程度：10%
    """
    
    score = 3.0  # 默认中等难度
    
    # 因子1: 文本复杂度
    text_score = self._calc_text_complexity(question_text)
    score += (text_score - 0.5) * 1.2  # 映射到 ±0.6
    
    # 因子2: 解题步骤
    if solution_steps is not None:
        step_score = min(solution_steps / 5.0, 1.0)  # 5步以上算满分
        score += (step_score - 0.5) * 1.0  # 映射到 ±0.5
    
    # 因子3: 耗时
    if time_spent is not None:
        # 假设正常耗时为30秒，以此为基准
        time_score = min(time_spent / 120.0, 1.0)  # 2分钟以上算满分
        score += (time_score - 0.5) * 0.8
    
    # 因子4: 工具使用
    if tools_used:
        tool_penalty = len([t for t in tools_used if t != "calculator"]) * 0.3
        score += tool_penalty
    
    # 因子5: 高级关键词检测
    advanced_keywords = [
        "泰勒展开", "傅里叶", "拉普拉斯", "偏微分",
        "多重积分", "格林公式", "斯托克斯",
    ]
    advanced_count = sum(1 for kw in advanced_keywords if kw in question_text)
    score += advanced_count * 0.4
    
    # 归一化到 1-5 范围
    final_score = max(1, min(5, round(score)))
    
    logger.debug(
        f"难度估算: raw={score:.1f} → final={final_score} "
        f"(text={text_score:.2f}, steps={solution_steps}, "
        f"time={time_spent}s, tools={len(tools_used) if tools_used else 0})"
    )
    
    return final_score


def _calc_text_complexity(self, text: str) -> float:
    """计算文本复杂度（0.0-1.0）"""
    if not text:
        return 0.0
    
    factors = []
    
    # 长度因子
    length_ratio = min(len(text) / 200.0, 1.0)
    factors.append(length_ratio * 0.4)
    
    # 数学符号密度
    math_chars = set("+−×÷=∫∑∏√∂∞πθαβγΔ∇±≤≥≈≠")
    math_count = sum(1 for c in text if c in math_chars)
    density = math_count / max(len(text), 1)
    factors.append(min(density * 10, 1.0) * 0.3)
    
    # 特殊结构（分数、上下标等暗示）
    special_patterns = [r"\^{", r"_{", r"\\frac", r"\\sqrt"]
    pattern_count = sum(1 for p in special_patterns if re.search(p, text))
    factors.append(min(pattern_count / 3.0, 1.0) * 0.3)
    
    return sum(factors) / len(factors) if factors else 0.0
```

##### 方法: 批量缓冲机制

```python
class _EventBuffer:
    """线程安全的事件缓冲区"""
    
    def __init__(self, max_size: int = 10):
        self._buffer: List[Dict[str, Any]] = []
        self._max_size = max_size
        self._lock = asyncio.Lock()
        self._last_flush_time = time.time()
    
    async def add(self, event: Dict[str, Any]) -> bool:
        async with self._lock:
            if len(self._buffer) >= self._max_size:
                logger.warning(f"缓冲区已满 ({self._max_size})，丢弃最新事件")
                return False
            
            self._buffer.append(event)
            return True
    
    async def drain(self) -> List[Dict[str, Any]]:
        async with self._lock:
            events = self._buffer.copy()
            self._buffer.clear()
            self._last_flush_time = time.time()
            return events
    
    @property
    def size(self) -> int:
        return len(self._buffer)
    
    @property
    def is_full(self) -> bool:
        return len(self._buffer) >= self._max_size
```

```python
async def _auto_flush_loop(self) -> None:
    """后台协程：定时检查并刷新缓冲区"""
    
    while self._running:
        try:
            await asyncio.sleep(self._flush_interval)
            
            should_flush = (
                self._buffer.is_full or
                (time.time() - self._buffer._last_flush_time) >= self._flush_interval
            )
            
            if should_flush and self._buffer.size > 0:
                logger.debug(
                    f"自动刷新触发: buffer_size={self._buffer.size}, "
                    f"interval={time.time() - self._buffer._last_flush_time:.1f}s"
                )
                
                result = await self._flush_buffer()
                
                if result["failed_count"] > 0:
                    logger.warning(
                        f"批量持久化部分失败: "
                        f"success={result['success_count']}, "
                        f"failed={result['failed_count']}"
                    )
                    
        except asyncio.CancelledError:
            logger.info("自动刷新循环已取消")
            break
        except Exception as e:
            logger.error(f"自动刷新循环异常: {e}", exc_info=True)
            await asyncio.sleep(1.0)  # 避免异常时忙等待
```

```python
async def _flush_buffer(self) -> Dict[str, Any]:
    """
    执行实际的批量写入数据库。
    
    包含重试机制和错误隔离（单条失败不影响其他）。
    """
    
    events = await self._buffer.drain()
    
    if not events:
        return {"success_count": 0, "failed_count": 0, "errors": []}
    
    success_count = 0
    failed_count = 0
    errors = []
    
    try:
        from app.data.repositories import LearningRecordRepository
        
        async with self._db_session_factory() as db:
            repo = LearningRecordRepository(db)
            
            for event in events:
                retry_count = 0
                last_error = None
                
                while retry_count < self._max_retries:
                    try:
                        await repo.create(**event)
                        success_count += 1
                        break
                        
                    except Exception as e:
                        last_error = e
                        retry_count += 1
                        
                        if retry_count < self._max_retries:
                            wait_time = RETRY_BACKOFF_BASE * (2 ** (retry_count - 1))
                            logger.warning(
                                f"持久化失败（第{retry_count}次重试）: {e}，"
                                f"{wait_time:.1f}s后重试"
                            )
                            await asyncio.sleep(wait_time)
                
                else:
                    # 所有重试都失败
                    failed_count += 1
                    errors.append({
                        "event_id": event.get("id", "unknown"),
                        "error": str(last_error),
                        "retries": retry_count,
                    })
                    
                    # 单条失败不回滚整个批次
                    logger.error(
                        f"事件持久化最终失败（已重试{retry_count}次）: {last_error}"
                    )
            
            # 显式提交事务
            await db.commit()
            
    except Exception as e:
        # 数据库连接级别的严重错误
        logger.error(f"数据库批量写入严重错误: {e}", exc_info=True)
        
        # 将所有事件标记为失败
        failed_count = len(events)
        errors = [{"error": str(e), "events_count": len(events)}]
        
        # 尝试重新放回缓冲区（下次再试）
        for event in events:
            await self._buffer.add(event)
    
    total = success_count + failed_count
    
    logger.info(
        f"批量持久化完成: total={total}, "
        f"success={success_count}, failed={failed_count}"
    )
    
    # 更新统计
    self._stats["total_persisted"] += success_count
    self._stats["total_failed"] += failed_count
    self._stats["last_flush_time"] = time.time()
    
    return {
        "success_count": success_count,
        "failed_count": failed_count,
        "errors": errors,
    }
```

```python
        await persistence_manager.persist_interaction(

            session_id=f"sess_{i}",

    assert result["success_count"] == 3
    assert result["failed_count"] == 0
    assert persistence_manager._buffer.size == 0
```

```python
@router.get("/api/health/persistence")
async def check_persistence_health(agent: MathAgent = Depends(get_agent)):
    """检查持久化系统健康状态"""
    if not agent._persistence_manager:
        return {"status": "disabled", "reason": "持久化未启用"}
    
    stats = agent.persistence_stats
    
    return {
        "status": "healthy" if stats["failed_rate"] < 0.05 else "degraded",
        "stats": stats,
        "recommendations": _generate_recommendations(stats),
    }


def _generate_recommendations(stats: Dict) -> List[str]:
    """基于统计数据生成运维建议"""
    recommendations = []
    
    if stats.get("failed_rate", 0) > 0.1:
        recommendations.append("⚠️ 失败率偏高，请检查数据库连接")
    
    if stats.get("buffer_size", 0) > 50:
        recommendations.append("⚠️ 缓冲区积压过多，考虑增大写入并发度")
    
    avg_interval = stats.get("avg_flush_interval", 0)
    if avg_interval > 10:
        recommendations.append("⚠️ 刷新间隔过长，可能存在性能瓶颈")
    
    return recommendations
```

## 七、风险与应对

### 7.1 技术风险

| 风险 | 概率 | 影响 | 应对措施 |
|------|------|------|---------|
| DB写入瓶颈 | 中 | 高 | 批量写入 + 连接池 + 异步IO |
| 分类不准确 | 中 | 中 | 两级策略 + 人工校验 + 反馈学习 |
| 内存泄漏（缓冲区） | 低 | 高 | 上限控制 + 定期检查 + 监控告警 |
| 并发竞争条件 | 低 | 中 | asyncio.Lock + 事务隔离 |
| LLM分类超时 | 低 | 低 | 超时控制 + 降级到关键词匹配 |

### 7.2 回滚方案

如果生产环境出现严重问题：

1. **配置开关**: `enable_memory_persistence=False` 立即禁用
2. **降级模式**: 仅保留关键词分类，禁用LLM分类
3. **同步模式**: 关闭异步，改为同步写入（牺牲性能保可靠）
