## 一、问题排查结果

### 1.1 问题一：数据存储机制缺陷

#### 1.1.1 问题描述
用户反馈：将题目添加到错题本后，打开错题本查看时发现**无法同时保存错题和答案**。

#### 1.1.2 根本原因定位

通过代码审查和数据文件分析，发现以下关键问题：

**数据存储现状**（基于 `data/error_book.json` 实际数据分析）：

| 字段 | 实际存储内容 | 预期内容 | 问题程度 |
|------|------------|---------|---------|
| `question` | `data:image/png;base64,iVBORw0K...` (base64图片编码) | 题目文本或图片路径 | **严重** |
| `question_type` | `"image"` | `"image"` 或 `"text"` | 正常 |
| `image_path` | `null` | 图片保存路径或base64数据 | **缺失** |
| `correct_answer` | 完整的答案文本（含LaTeX公式） | 答案内容 | **正常但未渲染** |
| `original_answer` | `""` (空字符串) | 用户原始错误答案 | **空缺** |
| `error_reason` | `"用户主动添加"` | 错误原因描述 | **过于简单** |

**核心问题**：
1. **题目字段存储了原始base64数据而非处理后的内容**
2. **image_path字段为null，导致无法正确显示图片**
3. **original_answer字段始终为空**
4. **数据验证机制缺失**

```python
def validate_error_book_data(data: dict) -> tuple[bool, list[str]]:
    """
    校验错题数据的完整性
    
    Returns:
        (是否有效, 错误信息列表)
    """
    errors = []
    
    # 必填字段检查
    if not data.get('question') or data['question'].strip() == '':
        errors.append("题目不能为空")
    
    if not data.get('correct_answer') or data['correct_answer'].strip() == '':
        errors.append("正确答案不能为空")
    
    if not data.get('error_reason') or data['error_reason'].strip() == '':
        errors.append("错误原因不能为空")
    
    # 数据格式检查
    if data.get('question_type') == 'image':
        question = data.get('question', '')
        if not (question.startswith('data:image/') or 
                os.path.exists(question)):
            errors.append("图片数据格式无效")
    
    return len(errors) == 0, errors
```

**Step 2: 数据格式化处理器**

```python
def format_error_book_data(raw_data: dict) -> dict:
    """
    格式化错题数据，确保数据一致性
    
    处理规则：
    1. 图片类型题目：分离图片数据和文本
    2. 文本类型题目：清理HTML标签和特殊字符
    3. 答案处理：生成预览文本
    4. 时间戳统一格式
    """
    formatted = raw_data.copy()
    
    # 处理题目字段
    question = formatted.get('question', '')
    question_type = formatted.get('question_type', 'text')
    
    if question_type == 'image' and question.startswith('data:image'):
        # 图片类型：分离base64数据和元信息
        formatted['raw_image_data'] = question
        formatted['has_image'] = True
        
        # 尝试从correct_answer中提取识别后的题目文本
        correct_answer = formatted.get('correct_answer', '')
        extracted_text = extract_recognized_text(correct_answer)
        formatted['recognized_text'] = extracted_text
        formatted['display_question'] = extracted_text[:100] + ("..." if len(extracted_text) > 100 else "")
    else:
        # 文本类型：清理格式
        formatted['display_question'] = clean_html_tags(question)
        formatted['has_image'] = False
    
    # 处理答案字段
    correct_answer = formatted.get('correct_answer', '')
    formatted['answer_preview'] = generate_preview(correct_answer)
    
    # 处理时间戳
    if not formatted.get('added_at'):
        formatted['added_at'] = datetime.now().strftime("%Y-%m-%d %H:%M")
    
    # 设置默认分类
    if not formatted.get('categories'):
        formatted['categories'] = auto_classify(formatted)
    
    return formatted


def extract_recognized_text(answer: str) -> str:
    """
    从AI回答中提取识别后的题目文本
    
    规则：
    1. 查找【题目】标记后的内容
    2. 提取"问题确认："后面的句子
    3. 如果都找不到，返回前200字符
    """
    import re
    
    # 模式1：【题目】标记
    pattern1 = r'【题目】\s*(.+?)(?:\n|$)'
    match1 = re.search(pattern1, answer)
    if match1:
        return match1.group(1).strip()
    
    # 模式2：问题确认标记
    pattern2 = r'问题确认[：:]\s*(.+?)(?:\n|$)'
    match2 = re.search(pattern2, answer)
    if match2:
        return match2.group(1).strip()
    
    # 默认：返回答案的前200字符作为题目
    return answer[:200]


def generate_preview(text: str, max_length: int = 150) -> str:
    """生成文本预览（去除LaTeX标记）"""
    import re
    
    # 移除LaTeX公式标记
    preview = re.sub(r'\$\$[\s\S]*?\$\$', '[公式]', text)
    preview = re.sub(r'\$[^$]+\$', '[公式]', preview)
    
    # 清理换行和多余空白
    preview = ' '.join(preview.split())
    
    if len(preview) > max_length:
        preview = preview[:max_length] + "..."
    
    return preview


def auto_classify(data: dict) -> List[str]:
    """根据内容自动分类"""
    text = data.get('question', '') + ' ' + data.get('correct_answer', '')
    
    keywords = {
        '极限': ['极限', 'lim', '收敛'],
        '导数': ['导数', '微分', 'f\'', 'dy/dx'],
        '积分': ['积分', '∫', '不定积分', '定积分'],
        '微分方程': ['微分方程', '差分方程', 'Δy'],
        '级数': ['级数', '∑', 'Σ', '无穷级数']
    }
    
    categories = []
    for category, words in keywords.items():
        if any(word in text for word in words):
            categories.append(category)
    
    return categories if categories else ['其他']
```

```python
@app.post('/api/error-book')
def add_error(request: ErrorItemRequest):
    try:
        # Step 1: 数据校验
        is_valid, errors = validate_error_book_data(request.dict())
        if not is_valid:
            raise HTTPException(status_code=400, detail=", ".join(errors))
        
        # Step 2: 数据格式化
        formatted_data = format_error_book_data(request.dict())
        
        # Step 3: 创建对象并保存
        error_item = ErrorItem(**formatted_data)
        error_id = error_book_manager.add(error_item)
        
        # Step 4: 返回完整数据（包含处理后的字段）
        return {
            "id": error_id,
            "status": "success",
            "message": "错题添加成功",
            "data": error_item.to_dict()
        }
        
    except HTTPException as e:
        raise e
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"服务器内部错误: {str(e)}")
```

```javascript
function openDetailModal(errorId) {
    const item = errorBook.find(e => e.id === errorId);
    if (!item) return;
    
    // 填充数据
    fillModalContent(item);
    
    // 显示模态框
    showModal();
    
    // 渲染数学公式（重要！）
    setTimeout(() => {
        renderMathInModal();
    }, 100);  // 等待DOM更新完成
}

// 渲染模态框中的数学公式
function renderMathInModal() {
    if (window.MathJax && window.MathJax.typesetPromise) {
        const answerBox = document.querySelector('.answer-box');
        if (answerBox) {
            MathJax.typesetPromise([answerBox]).catch((err) => console.log('MathJax error:', err));
        }
    }
}
```
