# 数学AI助手 - 错题本功能优化需求文档

## 一、项目背景

### 1.1 项目概述
- **项目名称**：数学AI助手（Math AI Assistant）
- **技术栈**：FastAPI + HTML/CSS/JavaScript + LangChain + Qwen大模型
- **核心功能**：数学问题解答、图片识别、多轮对话、错题本管理

### 2.2 当前问题诊断

#### 问题1：按钮是全局唯一的
```javascript
// showErrorBookButtons() 函数
const oldButtons = document.getElementById('error-book-buttons');
if (oldButtons) oldButtons.remove();  // 每次都移除旧按钮
```

**影响**：所有AI回复共享同一个按钮，无法区分是哪道题的按钮。

#### 问题2：点击后按钮消失
```javascript

cancelButton.onclick = hideErrorBookButtons;  // 按钮直接被删除
```

**影响**：用户无法回头修改之前的决定，无法知道哪些题已加入。

### 4.4 关键函数实现

#### generateErrorBookButtonHTML(message)
```javascript
function generateErrorBookButtonHTML(message) {
    const status = message.errorBookStatus || 'pending';
    const msgId = message.id;

    if (status === 'added') {
        return `
            <div class="error-book-action" data-msg-id="${msgId}">
                <button class="error-btn added" disabled>✓ 已加入错题本</button>
                <span class="error-time">已保存</span>
            </div>
        `;
    }

    if (status === 'skipped') {
        return `
            <div class="error-book-action" data-msg-id="${msgId}">
                <button class="error-btn skipped" disabled>⏭️ 已跳过</button>
            </div>
        `;
    }

    // pending 状态
    return `
        <div class="error-book-action" data-msg-id="${msgId}">
            <button class="error-btn add" onclick="handleAddToErrorBook('${msgId}')">
                📚 加入错题本
            </button>
            <button class="error-btn skip" onclick="handleSkipErrorBook('${msgId}')">
                ❌ 不加入
            </button>
        </div>
    `;
}
```

#### handleAddToErrorBook(messageId)
```javascript
async function handleAddToErrorBook(messageId) {
    const chat = chats.find(c => c.id === currentChatId);
    const aiMessage = chat.messages.find(m => m.id === messageId);

    // 找到关联的用户问题（上一条user消息）
    const msgIndex = chat.messages.findIndex(m => m.id === messageId);
    const userMessage = chat.messages[msgIndex - 1];

    // 构建错题数据
    const errorData = {
        question: userMessage.content,
        question_type: userMessage.type || 'text',
        correct_answer: aiMessage.content,
        error_reason: '用户主动添加',
        categories: [],
        notes: '',
        mastery_level: 3,
        is_mastered: false
    };

    try {
        // 调用后端API
        const response = await fetch('/api/error-book', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(errorData)
        });

        if (response.ok) {
            const result = await response.json();

            // 更新本地消息状态
            aiMessage.errorBookStatus = 'added';
            aiMessage.errorBookId = result.id;
            saveChats();

            // 局部刷新按钮
            updateMessageErrorButton(messageId, 'added');

            // 显示成功提示
            showToast('✅ 已成功加入错题本！', 'success');
        } else {
            throw new Error(await response.text());
        }
    } catch (error) {
        showToast('❌ 添加失败: ' + error.message, 'error');
    }
}
```

#### handleSkipErrorBook(messageId)
```javascript
function handleSkipErrorBook(messageId) {
    const chat = chats.find(c => c.id === currentChatId);

```

#### updateMessageErrorButton(messageId, status)
```javascript
function updateMessageErrorButton(messageId, status) {
    const actionDiv = document.querySelector(`.error-book-action[data-msg-id="${messageId}"]`);
    if (!actionDiv) return;

    // 找到对应的AI消息
    const chat = chats.find(c => c.id === currentChatId);
    const message = chat.messages.find(m => m.id === messageId);

    // 替换按钮HTML
    actionDiv.innerHTML = generateErrorBookButtonHTML(message);
}
```

#### showToast(message, type)
```javascript
function showToast(message, type = 'info') {
    // 移除已存在的toast
    const existingToast = document.querySelector('.toast-container');
    if (existingToast) existingToast.remove();

    // 创建toast容器
    const toast = document.createElement('div');
    toast.className = `toast-container toast-${type}`;
    toast.innerHTML = `
        <span class="toast-message">${message}</span>
        <button class="toast-close" onclick="this.parentElement.remove()">×</button>
    `;

    document.body.appendChild(toast);

    // 3秒后自动消失
    setTimeout(() => {
        if (toast.parentElement) {
            toast.remove();
        }
    }, 3000);
}
```

```javascript
// loadChats() 函数中处理
function loadChats() {
    const savedChats = localStorage.getItem('math_ai_chats');
    if (!savedChats) return [];

    const chats = JSON.parse(savedChats);

    // 迁移旧数据：为每条AI消息添加默认字段

            if (msg.sender === 'ai' && !msg.errorBookStatus) {
                msg.errorBookStatus = 'pending';
                msg.errorBookId = null;
            }
        });
    });

    return chats;
}
```
